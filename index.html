// Tambahkan fungsi khusus untuk pengujian lokal (Dev Mode) function devLogin(usernameOverride) { const sessionData = { id: usernameOverride.toLowerCase(), name: "Dimas (Dev Mode)", role: "Developer", // Atau "Staff" sesuai kebutuhan pengujian initials: "DM" }; // Simpan session langsung ke LocalStorage localStorage.setItem('beunaSession', JSON.stringify(sessionData)); // Alihkan halaman sesuai role if (sessionData.role.toLowerCase() === 'developer') { window.location.href = 'developer.html'; } else { window.location.href = 'crm.html'; } }
